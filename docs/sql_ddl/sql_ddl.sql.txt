-- public.audit_logs definition

-- Drop table

-- DROP TABLE public.audit_logs;

CREATE TABLE public.audit_logs (
	id bigserial NOT NULL,
	tenant_id int4 NULL,
	user_id int4 NULL,
	"action" varchar(20) NOT NULL,
	resource_type varchar(100) NULL,
	resource_id varchar(100) NULL,
	"method" varchar(10) NOT NULL,
	"path" varchar(500) NOT NULL,
	status_code int4 NULL,
	ip_address varchar(64) NULL,
	request_id varchar(64) NULL,
	details json NULL,
	created_at timestamp DEFAULT now() NOT NULL,
	CONSTRAINT pk_audit_logs PRIMARY KEY (id)
);
CREATE INDEX ix_audit_logs_created_at ON public.audit_logs USING btree (created_at);
CREATE INDEX ix_audit_logs_resource_type ON public.audit_logs USING btree (resource_type);
CREATE INDEX ix_audit_logs_tenant_id ON public.audit_logs USING btree (tenant_id);
CREATE INDEX ix_audit_logs_user_id ON public.audit_logs USING btree (user_id);


-- public.procedure_codes definition

-- Drop table

-- DROP TABLE public.procedure_codes;

CREATE TABLE public.procedure_codes (
	code varchar(20) NOT NULL,
	legacy_code varchar(20) NULL,
	description varchar(500) NOT NULL,
	category varchar(100) NOT NULL,
	default_fee numeric(10, 2) DEFAULT 0 NOT NULL,
	default_duration_minutes int4 NULL,
	requires_tooth bool DEFAULT false NOT NULL,
	requires_surface bool DEFAULT false NOT NULL,
	requires_quadrant bool DEFAULT false NOT NULL,
	requires_lab bool DEFAULT false NOT NULL,
	is_ortho bool DEFAULT false NOT NULL,
	billing_order varchar(10) NULL,
	recall_interval int4 NULL,
	recall_unit varchar(10) NULL,
	is_active bool DEFAULT true NOT NULL,
	created_at timestamp DEFAULT now() NOT NULL,
